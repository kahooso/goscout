[PINTOS]
sudo apt-get install gcc-multilib make perl qemu qemu-system
sudo ln /usr/bin/qemu-system-x86_64 /usr/bin/qemu
tar -zxf pintos.tar.gz
cd pintos/src/utils

sudo cp -f backtrace pintos pintos-mkdisk pintos-set-cmdline Pintos.pm pintos-gdb ../misc/gdb-macros /usr/bin/
sudo chmod 755 /usr/bin/backtrace /usr/bin/pintos /usr/bin/pintos-mkdisk /usr/bin/pintos-set-cmdline /usr/bin/pintos-gdb
sudo chmod 644 /usr/bin/Pintos.pm /usr/bin/gdb-macros
cd ../../../
cd pintos/src/threads
make

pintos --qemu -- -q run alarm-single

# В случае возникновения ошибки при запуске из-за битых байтов, в строчке, которая указана в выводе, заменяем битые байты на последовательность @^V
sudo nano /usr/bin/pintos

pintos --qemu -- -q run alarm-single


# После этого может возникнуть ошибка, связанная с несовместимости версий gcc. Далее инструкция для того, чтобы установить gcc-4.7.
[GCC]
objdump -j.data -s build/lib/stdio.o
sudo nano /etc/apt/sources.list
deb http://dk.archive.ubuntu.com/ubuntu/ xenial main
deb http://dk.archive.ubuntu.com/ubuntu/ xenial universe
sudo apt update
sudo apt install gcc-4.7
gcc --version
sudo rm /usr/bin/gcc
sudo ln /usr/bin/gcc-4.7 /usr/bin/gcc
gcc --version
make clean
make
objdump -j.data -s build/lib/stdio.o
pintos --qemu -- -q run alarm-single
